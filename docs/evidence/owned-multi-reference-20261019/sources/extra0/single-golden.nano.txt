resource struct Counter { value: int, active: bool }
fn update(view: &mut Counter) -> int {
 set view.value (+ view.value 3)
 set view.active (not view.active)
 return view.value
}
shadow update {
 let mut owner: Counter = Counter { value: 7, active: true }
 assert (== (update &mut owner) 10)
 assert (== owner.value 10)
 assert (not owner.active)
 let Counter { value, active } = owner
 assert (== value 10)
 assert (not active)
}
fn main() -> int {
 let mut owner: Counter = Counter { value: 12, active: false }
 assert (== (update &mut owner) 15)
 assert owner.active
 let Counter { value, active } = owner
 assert (== value 15)
 assert active
 return 0
}
shadow main { assert true }
