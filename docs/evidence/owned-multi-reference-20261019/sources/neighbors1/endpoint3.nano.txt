resource struct Counter { value: int, active: bool }
fn update(view: &mut Counter) -> int {
 set view.value (+ view.value -3)
 set view.active (not view.active)
 return view.value
}
shadow update { let mut owner: Counter = Counter { value: 7, active: false }
 assert (== (update &mut owner) 4)
 let Counter { value, active } = owner assert (== value 4) assert active }
fn main() -> int {
 let mut owner: Counter = Counter { value: 7, active: false }
 assert (== (update &mut owner) 4)
 assert owner.active
 assert (== (update &mut owner) 1)
 let Counter { value, active } = owner
 assert (== value 1) assert (not active)
 return 0
}
shadow main { assert true }
