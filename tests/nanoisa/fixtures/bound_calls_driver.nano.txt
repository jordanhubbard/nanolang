import "src_nano/compiler/nanoisa_codegen.nano"
import "src_nano/parser.nano"
import "src_nano/compiler/lexer.nano"
import "src_nano/compiler/module_bindings.nano"
from "src_nano/compiler/diagnostics.nano" import diag_list_new

fn main() -> int {
    let source: string = "fn dep_a_value() -> int { return 37 }\nfn dep_a_relay() -> int { return (value) }\nfn dep_a_ping() -> void { (println \"A\") }\nfn dep_b_value() -> int { return 12 }\nfn dep_b_relay() -> int { return (value) }\nfn dep_b_ping() -> void { (println \"B\") }\nfn main() -> int {\n    assert (== (values.relay) 37)\n    assert (== (other.relay) 12)\n    assert (== (alias) 37)\n    (values.ping)\n    (other.ping)\n    return 0\n}\n"
    let diagnostics: List<CompilerDiagnostic> = (diag_list_new)
    let tokens: List<LexerToken> = (tokenize_string source "bound.nano" diagnostics)
    let parser: Parser = (parse_program tokens (list_LexerToken_length tokens) "bound.nano")
    (mb_reset [1, 4, 7])
    (mb_add 0 "value" "dep_a_value")
    (mb_add 1 "value" "dep_b_value")
    (mb_add 2 "values.relay" "dep_a_relay")
    (mb_add 2 "other.relay" "dep_b_relay")
    (mb_add 2 "values.ping" "dep_a_ping")
    (mb_add 2 "other.ping" "dep_b_ping")
    (mb_add 2 "alias" "dep_a_value")
    let assembly: string = (nanoisa_emit_parser_nasm parser)
    assert (> (str_length assembly) 0)
    (print assembly)
    (mb_reset [])
    return 0
}
shadow main { assert (== (main) 0) }
