import "src_nano/compiler/nanoisa_codegen.nano"
import "src_nano/parser.nano"
import "src_nano/compiler/lexer.nano"
import "src_nano/compiler/module_bindings.nano"
from "src_nano/compiler/diagnostics.nano" import diag_list_new
extern fn get_argc() -> int
extern fn get_argv(index: int) -> string
extern fn file_read(path: string) -> string

fn main() -> int {
    if (!= (get_argc) 4) { return 0 }
    if (and (!= (get_argv 3) "raw") (and (!= (get_argv 3) "whole") (!= (get_argv 3) "program"))) { return 0 }
    let source: string = (file_read (get_argv 1))
    let first: int = (string_to_int (get_argv 2))
    let diagnostics: List<CompilerDiagnostic> = (diag_list_new)
    let tokens: List<LexerToken> = (tokenize_string source "shadows.nano" diagnostics)
    let parser: Parser = (parse_program tokens (list_LexerToken_length tokens) "shadows.nano")
    (mb_reset [])
    let mut assembly: string = ""
    if (== (get_argv 3) "whole") { set assembly (nanoisa_emit_parser_nasm parser) }
    else if (== (get_argv 3) "program") { set assembly (nanoisa_emit_program_nasm parser) }
    else { set assembly (nanoisa_emit_shadows_nasm parser first) }
    if (== assembly "") { (println (nanoisa_last_lowering_error)) return 1 }
    (print assembly)
    return 0
}
shadow main { assert (== (main) 0) }
