import "src_nano/compiler/nanoisa_codegen.nano"
import "src_nano/parser.nano"
import "src_nano/compiler/lexer.nano"
import "src_nano/compiler/module_bindings.nano"
from "src_nano/compiler/diagnostics.nano" import diag_list_new
extern fn get_argc() -> int
extern fn get_argv(index: int) -> string
extern fn file_read(path: string) -> string

fn main() -> int {
    if (!= (get_argc) 4) { return 1 }
    let mode: string = (get_argv 3)
    if (and (!= mode "raw") (!= mode "program")) { return 1 }
    let source: string = (file_read (get_argv 1))
    let diagnostics: List<CompilerDiagnostic> = (diag_list_new)
    let tokens: List<LexerToken> = (tokenize_string source "raw-map.nano" diagnostics)
    let parser: Parser = (parse_program tokens (list_LexerToken_length tokens) "raw-map.nano")
    (mb_reset [])
    let mut assembly: string = ""
    if (== mode "program") {
        set assembly (nanoisa_emit_parser_nasm parser)
    } else {
        set assembly (nanoisa_emit_shadows_nasm parser (string_to_int (get_argv 2)))
    }
    if (== assembly "") { (println (nanoisa_last_lowering_error)) return 1 }
    (print assembly)
    return 0
}
shadow main { assert (!= (main) 0) }
