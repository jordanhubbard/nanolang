# I retain the identical formatter and shadow from compiler_modular.nano and
# nanoc_integrated.nano at 4d38548d0 after retiring those legacy drivers.
fn phase_to_string(phase: int) -> string {
    if (== phase CompilerPhase.PHASE_LEXER) { return "lexer" }
    if (== phase CompilerPhase.PHASE_PARSER) { return "parser" }
    if (== phase CompilerPhase.PHASE_TYPECHECK) { return "typecheck" }
    if (== phase CompilerPhase.PHASE_NANOISA) { return "nanoisa" }
    if (== phase CompilerPhase.PHASE_BACKEND) { return "backend" }
    if (== phase CompilerPhase.PHASE_RUNTIME) { return "runtime" }
    return "unknown"
}
shadow phase_to_string {
    assert (== (phase_to_string CompilerPhase.PHASE_LEXER) "lexer")
    assert (== (phase_to_string CompilerPhase.PHASE_NANOISA) "nanoisa")
    assert (== (phase_to_string CompilerPhase.PHASE_BACKEND) "backend")
    assert (== (phase_to_string CompilerPhase.PHASE_RUNTIME) "runtime")
    assert (== (phase_to_string -1) "unknown")
}
